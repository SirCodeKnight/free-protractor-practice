# ------------------------------------------
# Default rules for all well-behaved bots
# ------------------------------------------
User-agent: *
Allow: /

# Prevent crawling of non-public / backend paths
Disallow: /admin/
Disallow: /api/

# Prevent crawling of URLs with query parameters (saves crawl budget & prevents duplicate content issues)
Disallow: /*?*
Disallow: /*&*

# Googlebot Specific Rules

Allow: /*.svg$
